- Date and site :: when and where the DVT or PE occurred :: text

- Treatment :: the anticoagulant treatment given, and its duration :: text
- Current anticoagulation :: current anticoagulant medication, or its absence :: presence
- Thrombophilia workup :: a thrombophilia screen or known clotting disorder, or its absence :: presence
- Recurrence :: a recurrent DVT or PE, or its absence :: presence
- IVC filter :: an IVC filter, or its absence :: presence
